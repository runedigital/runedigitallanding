// --- MAIN APP ---
export default function Showroom() {
  // LEVA CONTROLS (ADMIN / SELECTOR)
  const { mode } = useControls('RUNE // SHOWROOM', { mode: { options: Object.keys(ASSETS) } })
  
  const [viewMode, setViewMode] = useState('IDLE')
  const [activeColor, setActiveColor] = useState(ASSETS['DARK_MATTER'].defaultColor) 
  const [activeText, setActiveText] = useState('DARK_MATTER')
  
  // DETECT MODIFICATION
  const defaultColor = ASSETS[mode]?.defaultColor || '#ffffff'
  const defaultText = mode
  const isModified = (activeColor !== defaultColor) || (activeText !== defaultText)

  // RESET ON MODE CHANGE
  useEffect(() => {
    setActiveColor(ASSETS[mode].defaultColor)
    setActiveText(mode)
  }, [mode])

  useEffect(() => {
    const params = new URLSearchParams(window.location.search)
    if (params.get('status') === 'confirmed') setViewMode('SUCCESS')
  }, [])

  const handleCloseSuccess = () => {
    setViewMode('IDLE')
    window.history.replaceState({}, document.title, window.location.pathname)
  }

  const handleReset = () => {
    setActiveColor(defaultColor)
    setActiveText(defaultText)
  }

  return (
    <>
      <GlobalStyles />
      
      {/* SCENE LAYER */}
      <div style={{ position: 'absolute', top: 0, left: 0, width: '100%', height: '100%' }}>
        {mode === 'DARK_MATTER' && <DARK_MATTER primaryColor={activeColor} text={activeText} />}
        {mode === 'CRYSTALIX' && <DEEPTHINK1 primaryColor={activeColor} text={activeText} />}
        {mode === 'GATEWAY' && <GATEWAY primaryColor={activeColor} text={activeText} />}
        {mode === 'INFERNO' && <INFERNO primaryColor={activeColor} text={activeText} />}
        {mode === 'KINETIC' && <KINETIC primaryColor={activeColor} text={activeText} />}
        {mode === 'MERCURY' && <MERCURY primaryColor={activeColor} text={activeText} />}
      </div>

      {/* UI LAYER */}
      <div style={{ opacity: viewMode === 'IDLE' ? 1 : 0.3, transition: 'opacity 0.5s ease', pointerEvents: viewMode === 'IDLE' ? 'auto' : 'none' }}>
        <BrandWatermark />
        <InfoCard mode={mode} />
        
        {/* THE CONTROL DECK (ALWAYS VISIBLE IN IDLE) */}
        <ControlDeck 
          activeColor={activeColor} setColor={setActiveColor} 
          activeText={activeText} setText={setActiveText} 
          defaultColor={defaultColor} defaultText={defaultText}
          isModified={isModified}
        />

        <div style={{ position: 'fixed', bottom: '50px', right: '430px', zIndex: 100 }}>
          <button onClick={() => setViewMode('ACQUIRE')} className="rune-btn" style={{ background: 'rgba(0,0,0,0.8)', color: '#fff', border: '1px solid #fff', padding: '20px 60px', fontSize: '14px', letterSpacing: '3px', fontFamily: "'Oswald', sans-serif", fontWeight: 600, cursor: 'pointer' }}>
            ACQUIRE ASSET
          </button>
        </div>
      </div>

      {/* MODALS */}
      {viewMode === 'ACQUIRE' && (
        <AcquisitionMatrix 
          mode={mode} 
          onClose={() => setViewMode('IDLE')} 
          onConfigTier2={() => setViewMode('CONFIG_T2')} 
          onConfigTier3={() => setViewMode('CONFIG_T3')} 
          isModified={isModified}
          onReset={handleReset}
        />
      )}
      
      {viewMode === 'CONFIG_T2' && <ConfigTier2 mode={mode} onClose={() => setViewMode('IDLE')} activeColor={activeColor} setColor={setActiveColor} activeText={activeText} setText={setActiveText} />}
      {viewMode === 'CONFIG_T3' && <ConfigTier3 mode={mode} onClose={() => setViewMode('IDLE')} />}
      {viewMode === 'SUCCESS' && <SuccessScreen onClose={handleCloseSuccess} />}
      
      {/* HIDDEN LEVA */}
      <Leva collapsed={true} hidden={true} />
    </>
  )
}
